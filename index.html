SOLIDITY CONTRACT SNIPPET
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.24;
import {FHE, euint32, externalEuint32} from "@fhevm/solidity/lib/FHE.sol";
import {SepoliaConfig} from "@fhevm/solidity/config/ZamaConfig.sol";

// SalaryShield: 0x54CA5F9e453Db2bb9bfE7F83863CfC5125F9F4D9
// USDC:      0x4C424Dc42717EbB6c043c645845E425D4F40fCe0
// Network: Ethereum Sepolia (11155111)
contract SalaryShield is SepoliaConfig {
  mapping(address => euint32) private salaries;

  function addEmployee(
    address employee,
    externalEuint32 encSalary,
    bytes calldata proof
  ) external {
    euint32 salary = FHE.fromExternal(encSalary, proof);
    salaries[employee] = salary;
    FHE.allowThis(salary);
    FHE.allow(salary, employee);
  }

  // USDC token (Sepolia): 0x4C424Dc42717EbB6c043c645845E425D4F40fCe0
  IERC20 public constant USDC =
    IERC20(0x4C424Dc42717EbB6c043c645845E425D4F40fCe0); // USDC on Sepolia

  function viewMyPayout(bytes32 pubKey)
    external view returns (bytes memory) {
    return TFHE.reencrypt(salaries[msg.sender], pubKey);
  }

  function claimPayout() external {
    uint32 netPay = FHE.decrypt(salaries[msg.sender]);
    USDC.transfer(msg.sender, netPay * 1e6); // 6 decimals
  }
}
